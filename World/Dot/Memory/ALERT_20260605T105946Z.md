import os

class ConstraintCheck:
    PROTECTED_FILES = {'wisdom.txt', 'motion.md', 'SAM_PERSONALITY.md'}

    @classmethod
    def validate_operation(cls, file_path: str):
        abs_path = os.path.abspath(file_path)
        filename = os.path.basename(abs_path)
        if filename in cls.PROTECTED_FILES:
            raise PermissionError(f"Security Violation: Attempted modification of protected file {filename}")
